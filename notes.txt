  // 3. Query the server for the warning details
  const [warnings] = await db.execute('SHOW WARNINGS');
  
  // 4. Log or handle each warning
  warnings.forEach(warn => {
    console.error(`[MySQL Warning ${warn.Code}] Level: ${warn.Level} | Message: ${warn.Message}`);
  });
} else if (result.affectedRows === 0) {

} else {

}
